CREATE TABLE public.assignment_submissions (
  id integer NOT NULL DEFAULT nextval('assignment_submissions_id_seq'::regclass),
  assignment_id integer NOT NULL,
  student_id uuid NOT NULL,
  content text,
  submitted_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  grade numeric,
  feedback text,
  graded_by uuid,
  graded_at timestamp without time zone,
  CONSTRAINT assignment_submissions_pkey PRIMARY KEY (id),
  CONSTRAINT assignment_submissions_assignment_id_fkey FOREIGN KEY (assignment_id) REFERENCES public.classroom_assignments(id),
  CONSTRAINT assignment_submissions_student_id_fkey FOREIGN KEY (student_id) REFERENCES auth.users(id),
  CONSTRAINT assignment_submissions_graded_by_fkey FOREIGN KEY (graded_by) REFERENCES auth.users(id)
);
CREATE TABLE public.classroom_assignments (
  id integer NOT NULL DEFAULT nextval('classroom_assignments_id_seq'::regclass),
  classroom_id integer NOT NULL,
  title character varying NOT NULL,
  description text,
  material_id integer,
  due_date timestamp without time zone,
  points integer DEFAULT 0,
  created_by uuid NOT NULL,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT classroom_assignments_pkey PRIMARY KEY (id),
  CONSTRAINT classroom_assignments_classroom_id_fkey FOREIGN KEY (classroom_id) REFERENCES public.classrooms(id),
  CONSTRAINT classroom_assignments_material_id_fkey FOREIGN KEY (material_id) REFERENCES public.classroom_materials(id),
  CONSTRAINT classroom_assignments_created_by_fkey FOREIGN KEY (created_by) REFERENCES auth.users(id)
);
CREATE TABLE public.classroom_enrollments (
  id integer NOT NULL DEFAULT nextval('classroom_enrollments_id_seq'::regclass),
  classroom_id integer NOT NULL,
  student_id uuid NOT NULL,
  enrolled_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  status character varying DEFAULT 'active'::character varying CHECK (status::text = ANY (ARRAY['active'::character varying, 'inactive'::character varying, 'pending'::character varying]::text[])),
  created_at timestamp without time zone DEFAULT now(),
  CONSTRAINT classroom_enrollments_pkey PRIMARY KEY (id),
  CONSTRAINT classroom_enrollments_classroom_id_fkey FOREIGN KEY (classroom_id) REFERENCES public.classrooms(id),
  CONSTRAINT classroom_enrollments_student_id_fkey FOREIGN KEY (student_id) REFERENCES auth.users(id)
);
CREATE TABLE public.classroom_materials (
  id integer NOT NULL DEFAULT nextval('classroom_materials_id_seq'::regclass),
  classroom_id integer NOT NULL,
  pdf_id integer NOT NULL,
  title character varying NOT NULL,
  description text,
  assigned_by uuid NOT NULL,
  assigned_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  due_date date,
  is_required boolean DEFAULT false,
  order_index integer DEFAULT 0,
  available_for_students boolean DEFAULT true,
  CONSTRAINT classroom_materials_pkey PRIMARY KEY (id),
  CONSTRAINT classroom_materials_classroom_id_fkey FOREIGN KEY (classroom_id) REFERENCES public.classrooms(id),
  CONSTRAINT classroom_materials_pdf_id_fkey FOREIGN KEY (pdf_id) REFERENCES public.pdf_uploads(id),
  CONSTRAINT classroom_materials_assigned_by_fkey FOREIGN KEY (assigned_by) REFERENCES auth.users(id)
);
CREATE TABLE public.classroom_progress (
  id integer NOT NULL DEFAULT nextval('classroom_progress_id_seq'::regclass),
  student_id uuid NOT NULL,
  classroom_material_id integer NOT NULL,
  output_id integer,

  CONSTRAINT classroom_progress_pkey PRIMARY KEY (id),

  CONSTRAINT classroom_progress_output_id_fkey FOREIGN KEY (output_id) REFERENCES public.classroom_study_outputs(id)
);
CREATE TABLE public.classroom_study_outputs (

  CONSTRAINT classroom_study_outputs_pkey PRIMARY KEY (id),

CREATE TABLE public.classrooms (
  id integer NOT NULL DEFAULT nextval('classrooms_id_seq'::regclass),
  name character varying NOT NULL,
  description text,
  teacher_id uuid NOT NULL,
  subject character varying,
  grade_level character varying,
  is_active boolean DEFAULT true,
  join_code character varying NOT NULL UNIQUE,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  updated_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT classrooms_pkey PRIMARY KEY (id),
  CONSTRAINT classrooms_teacher_id_fkey FOREIGN KEY (teacher_id) REFERENCES auth.users(id),
  CONSTRAINT fk_teacher FOREIGN KEY (teacher_id) REFERENCES public.profiles(id)
);
CREATE TABLE public.comments (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  post_id bigint NOT NULL,
  user_id uuid NOT NULL,
  text text NOT NULL,
  created_at timestamp without time zone DEFAULT now(),
  CONSTRAINT comments_pkey PRIMARY KEY (id),
  CONSTRAINT comments_post_id_fkey FOREIGN KEY (post_id) REFERENCES public.posts(id),
  CONSTRAINT comments_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.profiles(id)
);
CREATE TABLE public.notifications (
  id integer NOT NULL DEFAULT nextval('notifications_id_seq'::regclass),
  user_id uuid NOT NULL,
  task_id integer,
  method character varying NOT NULL CHECK (method::text = ANY (ARRAY['email'::character varying::text, 'whatsapp'::character varying::text])),
  scheduled_at timestamp without time zone NOT NULL,
  sent boolean DEFAULT false,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT notifications_pkey PRIMARY KEY (id),
  CONSTRAINT notifications_task_id_fkey FOREIGN KEY (task_id) REFERENCES public.plan_tasks(id),
  CONSTRAINT notifications_user_id_fkey FOREIGN KEY (user_id) REFERENCES auth.users(id)
);

  file_name character varying NOT NULL,

CREATE TABLE public.plan_tasks (
  id integer NOT NULL DEFAULT nextval('plan_tasks_id_seq'::regclass),
  plan_id integer NOT NULL,
  title character varying NOT NULL,
  description text,
  due_date date,
  completed boolean DEFAULT false,
  output_id integer,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  priority text NOT NULL DEFAULT 'medium'::text CHECK (priority = ANY (ARRAY['low'::text, 'medium'::text, 'high'::text])),
  updated_at timestamp with time zone NOT NULL DEFAULT now(),
  CONSTRAINT plan_tasks_pkey PRIMARY KEY (id),
  CONSTRAINT plan_tasks_plan_id_fkey FOREIGN KEY (plan_id) REFERENCES public.study_plans(id),
  CONSTRAINT plan_tasks_output_id_fkey FOREIGN KEY (output_id) REFERENCES public.study_outputs(id)
);
CREATE TABLE public.posts (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  classroom_id integer NOT NULL,
  user_id uuid NOT NULL,
  content text NOT NULL,
  created_at timestamp without time zone DEFAULT now(),
  file_url text,
  material_id integer,
  material_type character varying CHECK (material_type::text = ANY (ARRAY['ai_generated'::character varying, 'pdf'::character varying, 'link'::character varying, 'text'::character varying]::text[])),
  CONSTRAINT posts_pkey PRIMARY KEY (id),
  CONSTRAINT posts_classroom_id_fkey FOREIGN KEY (classroom_id) REFERENCES public.classrooms(id),
  CONSTRAINT posts_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.profiles(id)
);

CREATE TABLE public.study_plans (
  id integer NOT NULL DEFAULT nextval('study_plans_id_seq'::regclass),
  user_id uuid NOT NULL,
  title character varying NOT NULL,
  description text,
  start_date date,
  end_date date,
  notify_by_email boolean DEFAULT false,
  notify_by_whatsapp boolean DEFAULT false,
  created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
  status text NOT NULL DEFAULT 'active'::text CHECK (status = ANY (ARRAY['active'::text, 'paused'::text, 'completed'::text, 'inactive'::text])),

  CONSTRAINT study_plans_user_id_fkey FOREIGN KEY (user_id) REFERENCES auth.users(id)

CREATE TABLE public.subscriptions (
  user_id uuid NOT NULL,
  mp_preapproval_id text,
  status text DEFAULT 'pending'::text CHECK (status = ANY (ARRAY['authorized'::text, 'paused'::text, 'cancelled'::text, 'pending'::text])),
  reason text,
  amount numeric,
  currency character varying,
  frequency integer,
  frequency_type text,
  next_payment_date timestamp with time zone,
  created_at timestamp with time zone DEFAULT now(),
  updated_at timestamp with time zone DEFAULT now(),
  metadata jsonb,
  cancelled_at timestamp with time zone,
  CONSTRAINT subscriptions_pkey PRIMARY KEY (user_id),
  CONSTRAINT subscriptions_user_id_fkey FOREIGN KEY (user_id) REFERENCES auth.users(id)
);
